# Run this cell AFTER all three models and both vectorizers have been fitted.
# This uses the original emotion labels preserved in `unique_emotions` in your notebook.

from pathlib import Path
import json
import joblib

artifact_dir = Path("model_artifacts")
artifact_dir.mkdir(exist_ok=True)

joblib.dump(bow_vectorizer, artifact_dir / "bow_vectorizer.joblib")
joblib.dump(nb_model, artifact_dir / "nb_bow_model.joblib")
joblib.dump(tfidf_vectorizer, artifact_dir / "tfidf_vectorizer.joblib")
joblib.dump(nb2_model, artifact_dir / "nb_tfidf_model.joblib")
joblib.dump(logistic_model, artifact_dir / "logistic_tfidf_model.joblib")

# In your notebook, emotion_numbers maps original emotion names to integer IDs.
# Reverse it so the app can display the original emotion name.
id_to_emotion = {int(class_id): str(emotion) for emotion, class_id in emotion_numbers.items()}
with open(artifact_dir / "emotion_labels.json", "w", encoding="utf-8") as f:
    json.dump(id_to_emotion, f, ensure_ascii=False, indent=2)

print("Saved model artifacts to:", artifact_dir.resolve())
print("Emotion labels:", id_to_emotion)
